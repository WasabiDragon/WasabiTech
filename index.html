// Keep the background page locked while a popup is open. const dialogs = $$("dialog"); function syncDialogScrollLock() { const openDialog = dialogs.find(dialog => dialog.open); document.body.classList.toggle( "dialog-open", Boolean(openDialog) ); } dialogs.forEach(dialog => { dialog.addEventListener("close", syncDialogScrollLock); dialog.addEventListener("cancel", () => { requestAnimationFrame(syncDialogScrollLock); }); }); // Redirect mouse-wheel scrolling to the open popup, // even when the pointer is outside the popup itself. document.addEventListener("wheel", event => { const openDialog = dialogs.find(dialog => dialog.open); if (!openDialog) return; const scrollArea = openDialog.querySelector( ".detail-card, .modal-card" ); if (!scrollArea) return; event.preventDefault(); scrollArea.scrollTop += event.deltaY; }, { passive: false }); // Keep the body locked when dialogs open. const dialogObserver = new MutationObserver(syncDialogScrollLock); dialogs.forEach(dialog => { dialogObserver.observe(dialog, { attributes: true, attributeFilter: ["open"] }); });
